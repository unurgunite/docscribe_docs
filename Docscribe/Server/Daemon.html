def setup_runtime_state
  @last_request_time = Process.clock_gettime(Process::CLOCK_MONOTONIC)
  @running = false
  @server = nil
  @started_at = Time.now
  @last_sig_hash = nil
end

  @config_path = config_path
  @idle_timeout = idle_timeout

def start
  load_dependencies
  setup_socket
  @running = true
  $PROGRAM_NAME = "docscribe server (#{Dir.pwd})"
  write_pid
  listen_loop
end
