def run(argv)
  opts = parse_options(argv)
  return 0 if opts[:help]

  conf = Docscribe::Config.load(opts[:config])
  conf = Docscribe::CLI::ConfigBuilder.build(conf, parse_cli_overrides(argv))

  puts conf.to_h.to_yaml
  0
end
