def run(argv)
  config_path, cmd = parse_args(argv)
  return warn(BANNER) || 1 unless cmd

  case cmd
  when 'start' then start_server(config_path)
  when 'stop' then stop_server(config_path)
  when 'status' then show_status(config_path)
  else warn(usage) || 1
  end
end
