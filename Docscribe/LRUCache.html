Overview

Bounded LRU cache with O(1) access and eviction. Used by Server::Daemon for file rewrite caching.

def [](key)
  val = @data[key]
  return nil unless val

  @data.delete(key)
  @data[key] = val
  val
end

def []=(key, val)
  @data.delete(key) if @data.key?(key)
  @data[key] = val
  @data.shift if @data.size > @max_size
end
